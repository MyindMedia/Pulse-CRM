#!/usr/bin/env bash
# Deploys workers/pulse-media to Cloudflare Workers with the pulse-media R2 bucket bound as BUCKET.
# Token and account id come from 1Password; nothing is written to disk or printed.
# Per-studio buckets (docs/R2-PER-ORG-BUCKETS.md): the Worker also reads studio media buckets
# through R2's S3 API, so it gets R2_ACCOUNT_ID + ORG_BUCKET_ENV as plain vars and the S3 key
# pair as Worker secrets (sent on stdin, never on the command line).
# Usage: scripts/r2/deploy-media-worker.sh [org-bucket-env, default prod]
set -euo pipefail
CF_TOKEN="$(op read 'op://Security/Cloudflare Pulse OS/Your API Token')"
CF_ACCT="$(op read 'op://Security/Cloudflare Pulse OS/Account ID')"
A="https://api.cloudflare.com/client/v4/accounts/$CF_ACCT/workers"
H="Authorization: Bearer $CF_TOKEN"
DIR="$(cd "$(dirname "$0")/../../workers/pulse-media" && pwd)"
ORG_ENV="${1:-prod}"
META='{"main_module":"index.js","compatibility_date":"2025-09-01","keep_bindings":["secret_text"],"bindings":[{"type":"r2_bucket","name":"BUCKET","bucket_name":"pulse-media"},{"type":"plain_text","name":"R2_ACCOUNT_ID","text":"'"$CF_ACCT"'"},{"type":"plain_text","name":"ORG_BUCKET_ENV","text":"'"$ORG_ENV"'"}]}'
curl -fsS -X PUT "$A/scripts/pulse-media" -H "$H" \
  -F "metadata=${META};type=application/json" \
  -F "index.js=@${DIR}/index.js;type=application/javascript+module;filename=index.js" \
  | python3 -I -c "import sys,json;d=json.load(sys.stdin);print('deployed',d.get('success'),d.get('errors'))"
curl -fsS -X POST "$A/scripts/pulse-media/subdomain" -H "$H" -H "Content-Type: application/json" -d '{"enabled":true}' \
  | python3 -I -c "import sys,json;d=json.load(sys.stdin);print('workers.dev enabled',d.get('success'),d.get('errors'))"
curl -fsS "$A/subdomain" -H "$H" | python3 -I -c "import sys,json;d=json.load(sys.stdin);print('URL: https://pulse-media.'+d['result']['subdomain']+'.workers.dev')"
# S3 key pair for reading studio buckets: same derivation as set-convex-env.sh (token id, sha256(token)).
# Swap for a read-only R2 token when one exists (see the doc).
TOKEN_ID="$(curl -fsS "https://api.cloudflare.com/client/v4/accounts/$CF_ACCT/tokens/verify" -H "$H" | python3 -I -c 'import sys,json;print(json.load(sys.stdin)["result"]["id"])')"
SECRET="$(printf %s "$CF_TOKEN" | shasum -a 256 | cut -d' ' -f1)"
put_secret() {
  printf %s "$2" | python3 -I -c 'import sys,json;print(json.dumps({"name":sys.argv[1],"text":sys.stdin.read(),"type":"secret_text"}))' "$1" \
    | curl -fsS -X PUT "$A/scripts/pulse-media/secrets" -H "$H" -H "Content-Type: application/json" --data @- \
    | python3 -I -c "import sys,json;d=json.load(sys.stdin);print('secret $1',d.get('success'),d.get('errors'))"
}
put_secret R2_ACCESS_KEY_ID "$TOKEN_ID"
put_secret R2_SECRET_ACCESS_KEY "$SECRET"
